cmake_minimum_required(VERSION 3.24)
project(csharp_models VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(CTest)
include(cmake/Dependencies.cmake)

add_library(lab_core src/syntax.cpp src/graph.cpp src/protocol.cpp src/cfg.cpp src/SSA.cpp)
target_include_directories(lab_core PUBLIC include)
target_link_libraries(lab_core PUBLIC tree_sitter_runtime nlohmann_json::nlohmann_json PRIVATE csharp_grammar)
if(MSVC)
  target_compile_options(lab_core PRIVATE /W4 /utf-8)
else()
  target_compile_options(lab_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(lab-model src/main.cpp)
target_link_libraries(lab-model PRIVATE lab_core)
# Keep the Windows CLI runnable without the compiler's DLL directory in PATH.
if(MINGW)
  target_link_options(lab-model PRIVATE -static)
endif()
if(BUILD_TESTING)
  add_executable(lab-tests tests/core_tests.cpp)
  target_link_libraries(lab-tests PRIVATE lab_core)
  if(MINGW)
    target_link_options(lab-tests PRIVATE -static)
  endif()
  add_test(NAME core COMMAND lab-tests "${CMAKE_CURRENT_SOURCE_DIR}/examples")
endif()
install(TARGETS lab-model RUNTIME DESTINATION bin)
